# Let's inspect all html files in the directory to add Open Graph meta tags across all of them (index.html, about.html, performance.html, gallery.html / events.html, contact.html) import glob html_files = glob.glob("*.html") print("Found HTML files:", html_files) og_meta_tags = ''' ''' for filename in html_files: with open(filename, "r", encoding="utf-8") as f: content = f.read() # Check if og:title already exists to avoid duplicate injection if "property=\"og:title\"" not in content: # Insert before content = content.replace("", f"{og_meta_tags}\n") with open(filename, "w", encoding="utf-8") as f: f.write(content) print(f"Added Open Graph tags to {filename}") else: print(f"Open Graph tags already present in {filename}")
